# 批注富文本与双层步进 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有答题与批注组件中，建立“批注气泡焦点步进 + 气泡内富文本片段步进”的双层放映模型，同时统一批注/关联建立流为“先选整句，再选下划线颜色”，并补齐右键揭示与右下角翻页按钮。

**Architecture:** 第一层步进保留在放映运行时中，由 ↑↓ 负责当前页的组件级焦点，在答题与批注页面里具体表现为批注气泡焦点切换，并且只有焦点序列耗尽后才翻页。第二层步进只存在于当前焦点批注气泡内部，由 ←→ 控制显式标注片段的揭示和回退，不自动串到下一个气泡。批注/关联锚点继续以整句下划线为主，但这些锚点本身不进入第二层步进。

**Tech Stack:** 原生 HTML / CSS / JavaScript，Node `--test`，JSDOM，既有 `slides-runtime.js` / `quiz-annotation-runtime.js` / `editor-rich-text.js` / `annotation-store.js`。

```javascript
function stepTopLevel(direction) {
  const strategy = getCurrentTopLevelStrategy();
  if (!strategy) return false;
  return direction === 'forward'
    ? strategy.forwardTopLevel()
    : strategy.backwardTopLevel();
}

function stepFocusedFragments(direction) {
  const strategy = getCurrentTopLevelStrategy();
  if (!strategy || typeof strategy.stepFragment !== 'function') return false;
  return strategy.stepFragment(direction);
}

document.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowDown') {
    e.preventDefault();
    if (!stepTopLevel('forward')) next();
  }
  if (e.key === 'ArrowUp') {
    e.preventDefault();
    if (!stepTopLevel('backward')) prev();
  }
  if (e.key === 'ArrowRight') {
    e.preventDefault();
    stepFocusedFragments('forward');
  }
  if (e.key === 'ArrowLeft') {
    e.preventDefault();
    stepFocusedFragments('backward');
  }
});
```

```javascript
function beginUnderlineAction(actionKind) {
  const selection = captureSentenceSelection();
  if (!selection) {
    showInlineSelectionHint('请先整句选中要建立批注或关联的文本');
    return;
  }

  pendingUnderlineAction = { actionKind, selection };
  openUnderlinePopover(selection.rect);
}

function commitUnderlineAction(color) {
  if (!pendingUnderlineAction) return;
  const anchor = wrapSelectionAsSentenceAnchor(pendingUnderlineAction.selection, color);

  if (pendingUnderlineAction.actionKind === 'create-note') {
    createNoteFromAnchor(anchor);
  } else {
    enterLinkingModeFromAnchor(anchor, pendingUnderlineAction.actionKind);
  }

  pendingUnderlineAction = null;
}
```

```javascript
function shouldShowFragmentToolbar(selection) {
  const bubble = selection.anchorNode && selection.anchorNode.parentElement
    ? selection.anchorNode.parentElement.closest('.qa-note-bubble.note-active')
    : null;
  return !!(bubble && !selection.isCollapsed);
}

function wrapSelectionAsFragment(range, formatType, value) {
  const wrapper = document.createElement('span');
  wrapper.setAttribute('data-fragment-step', 'true');
  wrapper.setAttribute('data-fragment-format', formatType);
  applyFragmentFormat(wrapper, formatType, value);
  wrapper.appendChild(range.extractContents());
  range.insertNode(wrapper);
  return wrapper;
}
```

```javascript
function revealFragmentManually(fragment) {
  fragment.classList.add('fragment-visible');
  fragment.dataset.fragmentManualReveal = 'true';
  markFragmentAsRevealed(fragment);
}

function hideFragmentOnBacktrack(fragment) {
  fragment.classList.remove('fragment-visible');
  fragment.removeAttribute('data-fragment-manual-reveal');
  markFragmentAsHidden(fragment);
}

document.addEventListener('contextmenu', (e) => {
  const fragment = e.target.closest('[data-fragment-step="true"]');
  if (!fragment) return;
  e.preventDefault();
  revealFragmentManually(fragment);
});
```

```javascript
function stripTransientFragmentState(root) {
  root.querySelectorAll('[data-fragment-step="true"]').forEach((fragment) => {
    fragment.classList.remove('fragment-visible', 'fragment-pending');
    fragment.removeAttribute('data-fragment-manual-reveal');
  });
}

function collectNoteContentForSave(el) {
  const temp = document.createElement('div');
  temp.innerHTML = el.innerHTML;
  stripTransientFragmentState(temp);
  return temp.innerHTML;
}
```

```javascript
function stepTopLevel(direction) {
  const strategy = getCurrentTopLevelStrategy();
  if (!strategy) return false;
  return direction === 'forward'
    ? strategy.forwardTopLevel()
    : strategy.backwardTopLevel();
}

function stepFocusedFragments(direction) {
  const strategy = getCurrentTopLevelStrategy();
  if (!strategy || typeof strategy.stepFragment !== 'function') return false;
  return strategy.stepFragment(direction);
}

document.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowDown') {
    e.preventDefault();
    if (!stepTopLevel('forward')) next();
  }
  if (e.key === 'ArrowUp') {
    e.preventDefault();
    if (!stepTopLevel('backward')) prev();
  }
  if (e.key === 'ArrowRight') {
    e.preventDefault();
    stepFocusedFragments('forward');
  }
  if (e.key === 'ArrowLeft') {
    e.preventDefault();
    stepFocusedFragments('backward');
  }
});
```

```javascript
function beginUnderlineAction(actionKind) {
  const selection = captureSentenceSelection();
  if (!selection) {
    showInlineSelectionHint('请先整句选中要建立批注或关联的文本');
    return;
  }

  pendingUnderlineAction = { actionKind, selection };
  openUnderlinePopover(selection.rect);
}

function commitUnderlineAction(color) {
  if (!pendingUnderlineAction) return;
  const anchor = wrapSelectionAsSentenceAnchor(pendingUnderlineAction.selection, color);

  if (pendingUnderlineAction.actionKind === 'create-note') {
    createNoteFromAnchor(anchor);
  } else {
    enterLinkingModeFromAnchor(anchor, pendingUnderlineAction.actionKind);
  }

  pendingUnderlineAction = null;
}
```

```javascript
function shouldShowFragmentToolbar(selection) {
  const bubble = selection.anchorNode && selection.anchorNode.parentElement
    ? selection.anchorNode.parentElement.closest('.qa-note-bubble.note-active')
    : null;
  return !!(bubble && !selection.isCollapsed);
}

function wrapSelectionAsFragment(range, formatType, value) {
  const wrapper = document.createElement('span');
  wrapper.setAttribute('data-fragment-step', 'true');
  wrapper.setAttribute('data-fragment-format', formatType);
  applyFragmentFormat(wrapper, formatType, value);
  wrapper.appendChild(range.extractContents());
  range.insertNode(wrapper);
  return wrapper;
}
```

```javascript
function revealFragmentManually(fragment) {

function hideFragmentOnBacktrack(fragment) {
  fragment.classList.remove('fragment-visible');
  fragment.removeAttribute('data-fragment-manual-reveal');
  markFragmentAsHidden(fragment);
}

document.addEventListener('contextmenu', (e) => {
```

```javascript
function stripTransientFragmentState(root) {
  root.querySelectorAll('[data-fragment-step="true"]').forEach((fragment) => {
    fragment.classList.remove('fragment-visible', 'fragment-pending');
    fragment.removeAttribute('data-fragment-manual-reveal');
  });
}

function collectNoteContentForSave(el) {
  const temp = document.createElement('div');
  temp.innerHTML = el.innerHTML;
  stripTransientFragmentState(temp);
  return temp.innerHTML;
}
```
